💡 EJEMPLO: Endpoint Seguro
════════════════════════════════════════════════════════════════════════════════

<?php
// api/reporte.php

session_start();
require_once __DIR__ . "/../config/db.php";
require_once __DIR__ . "/../config/session_security.php";

// 1. VALIDAR
requireAuth();
requireSessionContext();
$ctx = getSessionContext();

// 2. LEER
if ($_GET['action'] === 'get') {
    $id = (int)$_GET['id'];
    
    // Validar acceso
    if (!canAccessRecord($pdo, 'reporte', $id)) {
        respond(['error' => 'Forbidden'], 403);
    }
    
    // Query segura
    $query = "SELECT * FROM reporte WHERE id = :id 
             AND codigo_empresa = :empresa AND codigo_finca = :finca";
    $stmt = $pdo->prepare($query);
    $stmt->execute([
        ':id' => $id,
        ':empresa' => $ctx['codigo_empresa'],
        ':finca' => $ctx['codigo_finca']
    ]);
    
    $reporte = $stmt->fetch();
    respond(['ok' => true, 'data' => $reporte]);
}
